// Add the 24 restraint braces on the TSu-TSd mechanical interconnect.
// Everything else is identical to v08.
#include "Offline/Mu2eG4/geom/TransportSolenoid_v08.txt"

//-----------------------------------------------------------------------------
// TSu-TSd mechanical interconnect restraint braces ("Brace-12" design)
//
// Installed 2025 to hold the antiproton window housing rim at its corrected
// diameter. Sources: Mu2e-doc-54322 (repair summary) and Mu2e-doc-50662
// (heating and welding procedure, Figure 6 for the profile, Figure 7 for the
// layout). Each brace is an A-frame of 4" x 4" 304 stainless bars: two side
// bars welded to the TSu and TSd ring plates and a center bar above the
// housing rim. It is modeled as one extruded profile, 4" thick in the
// tangential direction. The block nut, threaded rod, nuts and washers
// (about 7 kg per brace) are not modeled: they sit on the housing rim, which
// this geometry models as a solid disk out to pbar.support.outROut = 1050 mm,
// while the as-built rim radius is about 1000 mm.
//-----------------------------------------------------------------------------
bool   ts.braces.build         = true;
string ts.braces.material      = "StainlessSteel";
// Radius of the brace feet: interconnect shell OD (ts.tsudinterconnect.rOut
// = 955.875 mm) plus 0.125 mm clearance.
double ts.braces.rBase         = 956.0;   // mm
double ts.braces.halfThickness = 50.8;    // mm, half of a 4" bar

// Profile in the plane that contains the TS3 axis, clockwise.
// profileAxial:  mm along the TS3 axis from the TS3 center (+ is toward TSu).
// profileRadial: mm radially outward from rBase.
// The feet at +-454 mm sit against the ring plates (inner faces at +-454.1 mm).
// Side bars rise at 18 deg; the center bar is 101.6 mm tall, its underside
// 121.24 mm above the feet, 161.71 mm long at the bottom and 193.90 mm at
// the top (ends cut at 99 deg). Gives 78 kg per brace with StainlessSteel.
vector<double> ts.braces.profileAxial  = { -454.0, -454.0,  -96.95,  96.95, 454.0,   454.0,  80.855, -80.855 };
vector<double> ts.braces.profileRadial = {    0.0, 106.829, 222.84, 222.84, 106.829,   0.0, 121.24,  121.24  };

// Azimuth of each brace around the TS3 axis, in degrees from +y (up) toward
// +z (east). One brace is placed per entry, in the order Brace-01 to Brace-24.
// Layout from drawing A-Frame-Brace-Assy-04 (J. Brandt, 01-Jul-2024), as
// printed in Mu2e-doc-50662 Figure 7 and Mu2e-doc-54322 slide 14. Braces 1-12
// are on the east side from bottom to top, 13-24 the mirror image on the west.
// The values follow the drawing's dimensions: the top brace of each side at
// 45 deg; 10 deg steps within each group of 3; 9 and 6 deg to the ports at
// +-74 deg, 6 deg to the ports at +-106 deg; 8.5 deg steps around the
// features at 106, 143 and 180 deg. They agree with the brace positions
// measured on the drawing to within 0.5 deg.
vector<double> ts.braces.phi = {
   171.5,  161.5,  151.5,    134.5,  124.5,  114.5,
   100.0,   90.0,   80.0,     65.0,   55.0,   45.0,
  -171.5, -161.5, -151.5,   -134.5, -124.5, -114.5,
  -100.0,  -90.0,  -80.0,    -65.0,  -55.0,  -45.0
};
